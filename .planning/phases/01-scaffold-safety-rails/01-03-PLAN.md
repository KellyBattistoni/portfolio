  <action>
Create `src/hooks/useDeviceCapabilities.ts`. No external dependencies — pure React + browser APIs.

**Four capability axes:**

1. `prefersReducedMotion` — reactive via MediaQueryList `change` event (WCAG best practice: must update when user toggles OS setting without page reload)
2. `isMobile` — `window.innerWidth <= 430` per locked decision, reactive via debounced resize listener (16ms debounce — one rAF frame)
3. `isLowEnd` — `hardwareConcurrency <= 4` (primary) OR `deviceMemory < 4` (secondary, Chromium-only — guard with `!== undefined` because Firefox/Safari don't implement it). Read once on mount (hardware doesn't change)
4. `supportsWebGL2` — create an offscreen canvas, attempt `getContext('webgl2')`, store boolean, discard canvas. Read once on mount

**Critical implementation notes:**
- `deviceMemory` is `undefined` on Firefox/Safari — always guard: `typeof navigator.deviceMemory !== 'undefined' && navigator.deviceMemory < 4`
- All window/navigator accesses must be guarded with `typeof window !== 'undefined'` for future SSR safety (even though this is Vite/CSR, it's a good habit and Phase 1 decision)
- Cleanup all listeners in useEffect return function to prevent memory leaks
- Initial state must be computed synchronously (not `false` then updated) to avoid a flash of wrong capability on first render

```ts
import { useState, useEffect } from 'react'

export interface DeviceCapabilities {
  prefersReducedMotion: boolean
  isMobile: boolean
  isLowEnd: boolean
  supportsWebGL2: boolean
}

function getInitialCapabilities(): DeviceCapabilities {
  if (typeof window === 'undefined') {
    return { prefersReducedMotion: false, isMobile: false, isLowEnd: false, supportsWebGL2: false }
  }

  const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches
  const isMobile = window.innerWidth <= 430

  // isLowEnd: hardwareConcurrency primary, deviceMemory secondary (Chromium only)
  const lowByCores = typeof navigator.hardwareConcurrency !== 'undefined'
    ? navigator.hardwareConcurrency <= 4
    : false
  const lowByMemory = typeof (navigator as Navigator & { deviceMemory?: number }).deviceMemory !== 'undefined'
    ? ((navigator as Navigator & { deviceMemory?: number }).deviceMemory ?? 8) < 4
    : false
  const isLowEnd = lowByCores || lowByMemory

  // supportsWebGL2: create offscreen canvas, test context, discard
  let supportsWebGL2 = false
  try {
    const canvas = document.createElement('canvas')
    supportsWebGL2 = !!canvas.getContext('webgl2')
  } catch {
    supportsWebGL2 = false
  }

  return { prefersReducedMotion, isMobile, isLowEnd, supportsWebGL2 }
}

export function useDeviceCapabilities(): DeviceCapabilities {
  const [capabilities, setCapabilities] = useState<DeviceCapabilities>(getInitialCapabilities)

  useEffect(() => {
    if (typeof window === 'undefined') return

    // 1. Reactive: prefersReducedMotion
    const mql = window.matchMedia('(prefers-reduced-motion: reduce)')
    const handleMotionChange = (e: MediaQueryListEvent) => {
      setCapabilities(prev => ({ ...prev, prefersReducedMotion: e.matches }))
    }
    mql.addEventListener('change', handleMotionChange)

    // 2. Reactive: isMobile (debounced to one rAF frame ~16ms)
    let resizeTimer: ReturnType<typeof setTimeout>
    const handleResize = () => {
      clearTimeout(resizeTimer)
      resizeTimer = setTimeout(() => {
        setCapabilities(prev => ({ ...prev, isMobile: window.innerWidth <= 430 }))
      }, 16)
    }
    window.addEventListener('resize', handleResize, { passive: true })

    // Cleanup
    return () => {
      mql.removeEventListener('change', handleMotionChange)
      window.removeEventListener('resize', handleResize)
      clearTimeout(resizeTimer)
    }
  }, []) // Run once on mount — isLowEnd and supportsWebGL2 don't need reactive updates

  return capabilities
}
```
