**src/components/error/AnimationErrorBoundary.tsx** — Use `react-error-boundary` v6 (NOT a hand-rolled class component). The fallback renders a static brand-gradient that looks like the Plasma backdrop's stand-in — not an error indicator:

```tsx
import { ErrorBoundary, type FallbackProps } from 'react-error-boundary'

function AnimationFallback({ error, resetErrorBoundary }: FallbackProps) {
  // Log to console; onError prop handles external reporting hookup in future phases
  console.error('[AnimationErrorBoundary] Caught error:', error)

  return (
    <div
      role="alert"
      style={{
        position: 'fixed',
        inset: 0,
        background: 'radial-gradient(ellipse at 50% 60%, rgba(255, 69, 0, 0.35) 0%, #050505 65%)',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        zIndex: 10,
      }}
    >
      {/* Intentionally minimal — looks like brand background, not an error */}
    </div>
  )
}

interface AnimationErrorBoundaryProps {
  children: React.ReactNode
  onError?: (error: Error, info: React.ErrorInfo) => void
}

export function AnimationErrorBoundary({ children, onError }: AnimationErrorBoundaryProps) {
  return (
    <ErrorBoundary
      FallbackComponent={AnimationFallback}
      onError={(error, info) => {
        console.error('[AnimationErrorBoundary] onError:', error, info)
        onError?.(error, info)
      }}
    >
      {children}
    </ErrorBoundary>
  )
}
```
